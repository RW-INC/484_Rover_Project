cmake_minimum_required(VERSION 3.15)
project(SPARX_Windows)

# Use C++17 for better performance/syntax
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)

# 1. Point CMake to your LibTorch folder BEFORE running find_package
# This tells CMake where to look for TorchConfig.cmake
set(CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/libtorch_cpu/libtorch")

# 2. Let LibTorch's CMake magic handle all the dependencies (Protobuf, C10, etc.)
find_package(Torch REQUIRED)

# 3. Find your source (.cpp) files
file(GLOB_RECURSE SOURCES 
    "Watchdog/*.cpp" 
    "Planning/*.cpp"
)

# 4. Find your headers (Optional for building, but required to see them in Visual Studio)
file(GLOB_RECURSE HEADERS 
    "Watchdog/*.hpp" 
    "Planning/*.hpp"
    "Utils/*.hpp"
)

# 5. Create the executable
add_executable(sparx_node ${SOURCES} ${HEADERS})

# 6. Include directories (You don't need the torch includes here anymore, find_package handles it)
target_include_directories(sparx_node PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/Watchdog
    ${CMAKE_CURRENT_SOURCE_DIR}/Planning
    ${CMAKE_CURRENT_SOURCE_DIR}/Utils
    ${CMAKE_CURRENT_SOURCE_DIR}/eigen
    ${CMAKE_CURRENT_SOURCE_DIR}/spline
)

# 7. Link LibTorch to your executable using the variable generated by find_package
target_link_libraries(sparx_node PRIVATE ${TORCH_LIBRARIES})

# (Windows specific) Ensure Torch links with the correct C++ standard
set_property(TARGET sparx_node PROPERTY CXX_STANDARD 17)
