cmake_minimum_required(VERSION 3.15)
project(SPARX_Windows)

set(CMAKE_BUILD_TYPE Release)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)

set(Torch_DIR "${CMAKE_CURRENT_SOURCE_DIR}/libtorch/share/cmake/Torch")
find_package(Torch REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")

file(GLOB_RECURSE SOURCES
    "Watchdog/*.cpp"
    "Planning/*.cpp"
    "Nav/*.cpp"
    "Comms/*.cpp"
    "Utils/*.cpp"
)

file(GLOB_RECURSE HEADERS
    "Watchdog/*.hpp"
    "Planning/*.hpp"
    "Nav/*.hpp"
    "Comms/*.hpp"
    "Utils/*.hpp"
)

add_executable(sparx_node ${SOURCES} ${HEADERS})

target_include_directories(sparx_node PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/Watchdog
    ${CMAKE_CURRENT_SOURCE_DIR}/Planning
    ${CMAKE_CURRENT_SOURCE_DIR}/Nav
    ${CMAKE_CURRENT_SOURCE_DIR}/Comms
    ${CMAKE_CURRENT_SOURCE_DIR}/Utils
    ${CMAKE_CURRENT_SOURCE_DIR}/eigen
    ${CMAKE_CURRENT_SOURCE_DIR}/spline
)

target_link_libraries(sparx_node PRIVATE ${TORCH_LIBRARIES})
